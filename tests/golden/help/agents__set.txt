usage: ag agents set [-h] [--worktree] [--base BASE]
                     [--backend {claude,opencode,gemini,codex,cursor,echo,api}] [--model MODEL]
                     [--dir ADIR] [--role {orchestrator,sub}] [--persona PERSONA]
                     [--system SYSTEM] [--instructions INSTRUCTIONS] [--mode MODE]
                     [--allow-cmd PATTERN] [--plan] [--no-plan] [--env K=V] [--env-unset K]
                     [--env-clear] [--claude-config-dir CLAUDE_CONFIG_DIR]
                     [--subscription-only {on,off}] [--transport {acp,cli}] [--acp-cmd ACP_CMD]
                     [--effort EFFORT] [--fallback FALLBACK] [--data DATA] [--stall-s STALL_S]
                     [--kit KIT] [--wake-require-diff {on,off}]
                     name

positional arguments:
  name

options:
  -h, --help            show this help message and exit
  --worktree            move agent into an isolated git worktree
  --base BASE
  --backend {claude,opencode,gemini,codex,cursor,echo,api}
  --model MODEL
  --dir ADIR
  --role {orchestrator,sub}
  --persona PERSONA     behavior preset from roles ("" clears)
  --system SYSTEM       inline custom system prompt ("" clears)
  --instructions INSTRUCTIONS
                        explicit .md guidance files (repeatable; "" clears)
  --mode MODE           permission mode ro|edits|auto|full ("" clears)
  --allow-cmd PATTERN   extra edits-mode shell allowlist prefix, opencode + claude (repeatable; ""
                        clears)
  --plan
  --no-plan
  --env K=V             child env var (repeatable; secret-looking keys need ${VAR}; PATH+=dir
                        prepends)
  --env-unset K         remove var from child env (repeatable)
  --env-clear           drop all per-agent env
  --claude-config-dir CLAUDE_CONFIG_DIR
                        CLAUDE_CONFIG_DIR for this agent (second account); clears claude session
                        on change
  --subscription-only {on,off}
                        strip API-key env (ANTHROPIC_*/OPENAI_*/GEMINI_/CURSOR_) so CLIs use
                        subscription login
  --transport {acp,cli}
  --acp-cmd ACP_CMD
  --effort EFFORT       reasoning effort, checked per model (highest|lowest resolve; "" clears):
                        claude --effort, opencode --variant, codex model_reasoning_effort
  --fallback FALLBACK   fallback when rate-limited: MODEL, or BACKEND/MODEL | BACKEND:MODEL to
                        fail over to another backend on a usage limit (repeatable, "" clears)
  --data DATA           data class work|public ("" clears to routing.json privacy.default)
  --stall-s STALL_S     stall watchdog seconds for this agent (0 = off; "" clears)
  --kit KIT             token-saving kit tools: codegraph,ctx,rtk,style | all | none ("" clears to
                        config kit.default; scoped MCP + style, ag kit doctor)
  --wake-require-diff {on,off}
                        wake diff gate default for this agent: fail wakes that change nothing in
                        the worktree (on|off; default on with --verify on an agent with a
                        worktree)
