usage: ag chat steer [-h] [--timeout TIMEOUT] [--force] name [text ...]

positional arguments:
  name
  text

options:
  -h, --help         show this help message and exit
  --timeout TIMEOUT  hard budget in seconds for the whole steer: queue/FIFO wait + the follow-up
                     turn (0=unlimited, default 30); steer on an idle agent just says so
  --force            bypass budget limits once (logged)
